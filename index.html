function シフト作成() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const raw = ss.getSheetByName("フォームの回答 1"); const output = ss.getSheetByName("整形"); const data = raw.getDataRange().getValues(); output.clear(); const names = ["A", "B", "C", "D", "E"]; // ヘッダー output.appendRow(["日付", ...names]); let monthDays = 31; for (let d = 1; d <= monthDays; d++) { let row = [d]; names.forEach(name => { let mark = ""; for (let i = 1; i < data.length; i++) { if (data[i][1] == name) { let days = data[i][3]; if (days && days.includes(d.toString())) { mark = "○"; } } } row.push(mark); }); output.appendRow(row); } }
